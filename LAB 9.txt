4
SELECT 
    e.LAST_NAME,
    p.PRODUCT_NAME,
    op.SELL_PRICE,
    COUNT(*) OVER (PARTITION BY p.PRODUCT_ID, CAST(o.ORDER_DATE AS DATE)
    ) AS ILOŚĆ_TRANSAKCJI_W_DNIU,
    SUM(op.SELL_PRICE * op.AMOUNT) OVER (PARTITION BY p.PRODUCT_ID, CAST(o.ORDER_DATE AS DATE)
    ) AS PŁATNOŚĆ_ZA_PRODUKT_DNIA,
    LAG(op.SELL_PRICE) OVER (PARTITION BY p.PRODUCT_ID 
        ORDER BY o.ORDER_DATE) AS POPRZEDNIA_CENA,
    LEAD(op.SELL_PRICE) OVER (PARTITION BY p.PRODUCT_ID 
        ORDER BY o.ORDER_DATE) AS NASTĘPNA_CENA
FROM ORDER_POSITION op
JOIN `ORDER` o ON o.ORDER_ID = op.ORDER_ID
JOIN EMPLOYEE e ON e.EMPLOYEE_NAME = o.EMPLOYEE_ID
JOIN PRODUCT p ON p.PRODUCT_ID = op.PRODUCT_ID;

5
SELECT 
    p.PRODUCT_NAME,
    op.SELL_PRICE,
    EXTRACT(MONTH FROM o.ORDER_DATE) AS MIESIĄC,
    SUM(op.SELL_PRICE * op.AMOUNT) 
	OVER (PARTITION BY EXTRACT(MONTH FROM o.ORDER_DATE)) AS SUMA_MIESIĘCZNA,
    SUM(op.SELL_PRICE * op.AMOUNT)
	OVER (PARTITION BY EXTRACT(MONTH FROM o.ORDER_DATE)ORDER BY o.ORDER_DATE) AS TERAŹNIEJSZA_SUMA_MIESIĘCZNA
FROM ORDER_POSITION op
JOIN `ORDER` o ON o.ORDER_ID = op.ORDER_ID
JOIN PRODUCT p ON p.PRODUCT_ID = op.PRODUCT_ID;


6
SELECT
    c.CATEGORY_NAME,
    SUM(CASE 
            WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 1 
            THEN op.SELL_PRICE 
        END) AS STYCZEŃ,
    SUM(CASE 
            WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 2 
            THEN op.SELL_PRICE 
        END) AS LUTY,
    SUM(CASE 
            WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 3 
            THEN op.SELL_PRICE 
        END) AS MARZEC,
    SUM(CASE 
            WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 4 
            THEN op.SELL_PRICE 
        END) AS KWIECIEŃ,
    ABS(SUM(CASE WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 4 THEN op.SELL_PRICE END)
	-SUM(CASE WHEN EXTRACT(MONTH FROM o.ORDER_DATE) = 1 THEN op.SELL_PRICE END)) AS RÓŻNICA_CEN
FROM ORDER_POSITION op
JOIN PRODUCT p 
    ON p.PRODUCT_ID = op.PRODUCT_ID
JOIN CATEGORY c 
    ON c.CATEGORY_ID = p.CATEGORY_ID
JOIN `ORDER` o 
    ON o.ORDER_ID = op.ORDER_ID
WHERE o.ORDER_DATE BETWEEN '2025-01-01' AND '2025-04-30'
GROUP BY c.CATEGORY_NAME;

7
SELECT
    c.CATEGORY_NAME,
    p.PRODUCT_NAME,
    p.PURCHASE_PRICE,

    MIN(p.PURCHASE_PRICE) OVER (PARTITION BY p.CATEGORY_ID) AS MIN_CENA,
    MAX(p.PURCHASE_PRICE) OVER (PARTITION BY p.CATEGORY_ID) AS MAX_CENA,

    MAX(p.PURCHASE_PRICE) OVER (PARTITION BY p.CATEGORY_ID) -
    MIN(p.PURCHASE_PRICE) OVER (PARTITION BY p.CATEGORY_ID) AS RÓŻNICA_CEN

FROM PRODUCT p
JOIN CATEGORY c ON c.CATEGORY_ID = p.CATEGORY_ID;

8
SELECT
    p.PRODUCT_NAME,
    op.SELL_PRICE,
    o.ORDER_DATE,

    AVG(op.SELL_PRICE) OVER (
        PARTITION BY p.PRODUCT_ID
        ORDER BY o.ORDER_DATE
        ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
    ) AS ŚREDNIA_KROCZĄCA

FROM ORDER_POSITION op
JOIN `ORDER` o 
    ON o.ORDER_ID = op.ORDER_ID
JOIN PRODUCT p 
    ON p.PRODUCT_ID = op.PRODUCT_ID
ORDER BY p.PRODUCT_NAME, o.ORDER_DATE;
