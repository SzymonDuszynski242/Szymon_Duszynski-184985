2.1
SELECT DEPARTMENT_ID, SUM(pensja) AS suma FROM EMPLOYEE GROUP BY DEPARTMENT_ID ORDER BY suma DESC;

2.2
SELECT JOB_POSITION_ID, AVG(pensja) AS srednia,
 COUNT(*) AS employee_count FROM EMPLOYEE GROUP BY JOB_POSITION_ID HAVING COUNT(*) >= 3 AND srednia > 3000;

2.3
SELECT DEPARTMENT_ID, COUNT(DISTINCT pensja) AS inna,
 COUNT(EMPLOYEE_NAME) FROM EMPLOYEE GROUP BY DEPARTMENT_ID HAVING COUNT(*) > 4 ORDER BY inna ASC;

3.1
CREATE TABLE DEPARTMENT AS SELECT * FROM company_2025.DEPARTMENT;
CREATE TABLE JOB_POSITION AS SELECT * FROM company_2025.JOB_POSITION;

3.2
SELECT d.DEPARTMENT_NAME, COUNT(e.EMPLOYEE_NAME) AS employee_count
FROM DEPARTMENT d LEFT JOIN EMPLOYEE e USING (DEPARTMENT_ID) GROUP BY d.DEPARTMENT_NAME;

3.3
SELECT d.DEPARTMENT_NAME, GROUP_CONCAT(CONCAT(e.FIRST_NAME, ' ', e.LAST_NAME) ORDER BY e.LAST_NAME SEPARATOR ', ') 
 AS employees FROM DEPARTMENT d JOIN EMPLOYEE e USING (DEPARTMENT_ID) GROUP BY d.DEPARTMENT_NAME;

4.1
SELECT FIRST_NAME, LAST_NAME, POSITION_NAME FROM EMPLOYEE NATURAL JOIN JOB_POSITION WHERE YEAR(BIRTH_DATE) BETWEEN 1990 AND 1999;

4.2
SELECT FIRST_NAME, LAST_NAME, BIRTH_DATE FROM EMPLOYEE e
JOIN JOB_POSITION j USING (JOB_POSITION_ID) WHERE j.POSITION_NAME = 'Magazynier' ORDER BY BIRTH_DATE DESC LIMIT 5;

4.3
SELECT  c1.SHORT_NAME AS client1, c2.SHORT_NAME AS client2 FROM CLIENT c1 JOIN CLIENT c2 
ON ABS(CAST(c1.CLIENT_ID as SIGNED) - CAST(c2.CLIENT_ID as SIGNED))= 5 ORDER BY c1.CLIENT_ID;

5.1
SELECT JOB_POSITION_ID, AVG(pensja) AS srednia, COUNT(*) AS ilosc
FROM EMPLOYEE e JOIN DEPARTMENT d USING (DEPARTMENT_ID)
WHERE d.DEPARTMENT_NAME NOT IN ('IT', 'HR') GROUP BY JOB_POSITION_ID HAVING COUNT(*) > 3;
