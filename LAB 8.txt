SELECT  c.full_name, a.street,
LENGTH(a.street) AS street_length,LEFT(a.street, 5) AS street_prefix FROM CLIENT c
JOIN CLIENT_ADDRESS a ON a.client_id = c.client_id;

1.2

c.short_name, a.address_type
FROM CLIENT c
LEFT JOIN CLIENT_ADDRESS a
ON a.client_id = c.CLIENT_ID;

1.3
SELECT 
p.PRODUCT_NAME,
cat.CATEGORY_NAME,
LEFT(p.PRODUCT_NAME, 3) AS product_prefix
FROM PRODUCT p
JOIN CATEGORY cat ON cat.CATEGORY_ID = p.category_id;

1.4

p.PRODUCT_NAME,
p.DESCRIPTION
FROM PRODUCT p
JOIN ORDER_POSITION op ON op.PRODUCT_ID= p.PRODUCT_ID

2.1
SELECT 
c.full_name,
o.order_number,
o.order_date,
DATEDIFF(CURDATE(), o.order_date) AS days_since_order
FROM `ORDER` o
JOIN CLIENT c ON c.CLIENT_ID = o.CLIENT_ID;

2.2
SELECT
o.ORDER_ID,
o.ORDER_DATE,
DATE_ADD(o.ORDER_ID, INTERVAL 14 DAY) AS payment_due_date
FROM `ORDER` o
WHERE o.ORDER_STATUS NOT IN ('Zwrocone', 'Zamkniete');

2.3
SELECT 
e.FIRST_NAME,
e.LAST_NAME,
e.DATE_OF_EMPLOYMENT,
DAYNAME(e.DATE_OF_EMPLOYMENT) AS hire_day,
d.DEPARTMENT_NAME AS department
FROM EMPLOYEE e
LEFT JOIN DEPARTMENT d ON d.DEPARTMENT_ID = e.department_id;

2.4
SELECT 
c.FULL_NAME,
DATEDIFF(MAX(o.ORDER_DATE), MIN(o.ORDER_DATE)) AS days_diff
FROM CLIENT c
LEFT JOIN `ORDER` o 
ON o.client_id = c.CLIENT_ID
GROUP BY c.CLIENT_ID, c.FULL_NAME;

3.1
SELECT 
c.SHORT_NAME,
GROUP_CONCAT(a.street SEPARATOR ', ') AS addresses
FROM CLIENT c
LEFT JOIN CLIENT_ADDRESS a 
ON a.CLIENT_ID = c.CLIENT_ID
GROUP BY c.CLIENT_ID, c.SHORT_NAME;

3.2
SELECT 
CITY,
GROUP_CONCAT(DISTINCT LEFT(postal_code, 2) SEPARATOR ', ') AS prefixes
FROM CLIENT_ADDRESS
GROUP BY city;

3.3
SELECT 
CONCAT(e.LAST_NAME, ' ', e.FIRST_NAME) AS employee,
d.DEPARTMENT_NAME AS department,
DATEDIFF(CURDATE(), e.DATE_OF_EMPLOYMENT) AS days_from_hire
FROM EMPLOYEE e
LEFT JOIN DEPARTMENT d ON d.DEPARTMENT_ID = e.DEPARTMENT_ID;

3.4
SELECT 
CONCAT(c.FULL_NAME, ' - ', c.SHORT_NAME) AS client,
GROUP_CONCAT(RIGHT(o.order_number, 3) SEPARATOR ', ') AS order_numbers
FROM CLIENT c
LEFT JOIN `ORDER` o 
ON o.CLIENT_ID = c.CLIENT_ID
GROUP BY 
c.CLIENT_ID, c.FULL_NAME, c.SHORT_NAME;

3.5
SELECT
  CONCAT(
LOWER(LEFT(FIRST_NAME, 3)),
UPPER(RIGHT(LAST_NAME, 4)),
LENGTH(FIRST_NAME) + LENGTH(YEAR(BIRTH_DATE)) + LENGTH(MONTH(BIRTH_DATE)) + LENGTH(DAY(BIRTH_DATE))
  ) AS HASŁO
FROM EMPLOYEE;


4.1
SELECT
DATE_FORMAT(o.order_date, '%d-%m-%Y') AS dmr,
DATE_FORMAT(o.order_date, '%Y-%u') AS rok_tydzień,
DATE_FORMAT(o.order_date, '%Y-%j') AS rok_dzień,
DATE_FORMAT(o.order_date, '%M/%d/%Y') AS english_name
FROM `ORDER` o;

4.2
SELECT 
WEEK(o.order_date, 1) AS week_number,
SUM(op.AMOUNT * op.SELL_PRICE) AS total_value
FROM `ORDER` o
JOIN ORDER_POSITION op ON op.ORDER_ID = o.ORDER
WHERE YEAR(o.ORDER_DATE) = 2025
GROUP BY WEEK(o.ORDER_DATE, 1)
ORDER BY total_value DESC
LIMIT 1;
