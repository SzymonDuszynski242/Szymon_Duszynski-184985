FROM CLIENT

FROM CLIENT c
LEFT JOIN `ORDER` o ON c.CLIENT_ID = o.CLIENT_ID
WHERE o.ORDER_ID IS NULL;

SELECT u.UNIT_NAME FROM INVENTORY i
 RIGHT JOIN UNIT u ON i.UNIT_ID = u.UNIT_ID
WHERE i.UNIT_ID IS NULL;

1.3
SELECT p.PRODUCT_NAME, AVG(op.SELL_PRICE - p.PURCHASE_PRICE) AS srednia
FROM ORDER_POSITION op
JOIN PRODUCT p ON p.PRODUCT_ID = p.PRODUCT_ID
WHERE op.SELL_PRICE IS NOT NULL
GROUP BY p.PRODUCT_NAME
ORDER BY srednia DESC;

2.1
SELECT DISTINCT d.DEPARTMENT_NAME FROM `ORDER` o 
JOIN EMPLOYEE e ON e.EMPLOYEE_NAME = o.EMPLOYEE_ID
JOIN DEPARTMENT d ON d.DEPARTMENT_ID = e.DEPARTMENT_ID 

2.2
SELECT d.DEPARTMENT_NAME,
SUM( i.AMOUNT * op.SELL_PRICE) AS SUMA
FROM `ORDER` o
JOIN ORDER_POSITION op ON o.ORDER_ID = op.ORDER_ID
JOIN EMPLOYEE e ON o.EMPLOYEE_ID = e.EMPLOYEE_NAME
JOIN DEPARTMENT d ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
Join INVENTORY i ON op.PRODUCT_ID = i.PRODUCT_ID
GROUP BY d.DEPARTMENT_NAME

2.3
SELECT d.DEPARTMENT_NAME,
ORDER_STATUS, 
SUM( i.AMOUNT * op.SELL_PRICE) AS SUMA
FROM `ORDER` o
JOIN ORDER_POSITION op ON o.ORDER_ID = op.ORDER_ID
JOIN EMPLOYEE e ON o.EMPLOYEE_ID = e.EMPLOYEE_NAME
JOIN DEPARTMENT d ON e.DEPARTMENT_ID = d.DEPARTMENT_ID
Join INVENTORY i ON op.PRODUCT_ID = i.PRODUCT_ID
GROUP BY d.DEPARTMENT_NAME, ORDER_STATUS

SELECT 
    CONCAT_WS(',', 
        p.PRODUCT_ID, 
        p.Product_name, 
        p.Description, 
        p.Purchase_Price, 
        op.Sell_Price ) AS CsvData
FROM PRODUCT p
JOIN ORDER_POSITION op ON p.PRODUCT_ID = op.Product_Id;
3.3
SELECT 
    d.Department_name AS DepartmentName,
    GROUP_CONCAT(CONCAT(e.FIRST_NAME, ' ', e.LAST_NAME), ', ') AS EmployeesList
FROM DEPARTMENT d
JOIN EMPLOYEE e ON d.DEPARTMENT_Id = e.Department_Id
GROUP BY d.Department_name;
