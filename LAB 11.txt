CREATE TABLE student (id_studenta INT PRIMARY KEY,imie VARCHAR(100) NOT NULL, nazwisko VARCHAR(100) NOT NULL, rok_studiow INT UNSIGNED, data_urodzenia DATE);

2
CREATE TABLE kierunek (id_kierunku INT UNSIGNED PRIMARY KEY, nazwa_kierunku VARCHAR(200) NOT NULL);


3
CREATE TABLE student_na_kierunku(student INT, FOREIGN KEY (student) REFERENCES student(id_studenta), kierunek INT, FOREIGN KEY (kierunek) REFERENCES kierunek(id_klienta))

4
CREATE TABLE przedmiot (id_przedmiotu INT AUTO_INCREMENT PRIMARY KEY, nazwa_przedmiotu VARCHAR(200) NOT NULL, opis TEXT);

5
CREATE TABLE kierunek_has_przedmioty 
(kierunek INT, przedmiot INT, obowiazkowy BOOLEAN DEFAULT 1,
FOREIGN KEY (kierunek) REFERENCES kierunek(id_klienta), FOREIGN KEY (przedmiot) REFERENCES przedmiot(id_przedmiotu));

SELECT s.indeks, s.imie, s.nazwisko, k.nazwa_kierunku, khp.semestr, p.nazwa_przedmiotu
FROM student s JOIN student_na_kierunku snk ON s.indeks = snk.student 
JOIN kierunek k ON snk.kierunek = k.id_klienta
JOIN kierunek_has_przedmioty khp ON k.id_klienta = khp.kierunek
JOIN przedmiot p ON khp.przedmiot = p.id_przedmiotu;
